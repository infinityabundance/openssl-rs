//! ## This file does not compile, on purpose
//!

/// `static void pem_free(void *p, unsigned int flags, size_t num, const char *file, int line)`
/// — `crypto/pem/pem_lib.c:225-232`, the body behind the `PEM_FREE(p, flags, num)` macro at
/// `:223`.
///
/// The `flags` word picks the allocator *and* the free: `PEM_FLAG_SECURE` means the block came
/// from `CRYPTO_secure_malloc` and must go back through `CRYPTO_secure_clear_free` **with its
/// length**, and anything else means `CRYPTO_free`, which ignores `num`. A caller that passes
/// `PEM_FLAG_SECURE` and then frees with a different `num` has changed what the allocator is told.
///
/// # Safety
///
/// `p` must be NULL or a block obtained from `CRYPTO_malloc`/`CRYPTO_secure_malloc` with the same
/// `flags`, and `num` must be that block's length when the secure flavour is selected.
unsafe fn pem_free(p: *mut c_void, flags: c_uint, num: usize, file: *const c_char, line: c_int) {
    if flags & PEM_FLAG_SECURE != 0 {
        // SAFETY: the caller's contract: the block came from the secure allocator with this
        // length.
        unsafe { CRYPTO_secure_clear_free(p, num, file, line) }
    } else {
        // SAFETY: the caller's contract, and `CRYPTO_free` reads only the pointer.
        unsafe { CRYPTO_free(p, file, line) }
    }
}

/// `static int check_pem(const char *nm, const char *name)` — `crypto/pem/pem_lib.c:128-221`.
///
/// `nm` is the name a PEM header actually carried and `name` is the one the caller asked for;
/// this answers whether they are the same kind of block. It is an exact `strcmp` first, then a
/// list of one-way aliases the authority has accumulated since 0.9.x, in a fixed order:
///
/// * `ANY PRIVATE KEY` matches `ENCRYPTED PRIVATE KEY`, `PRIVATE KEY`, and any name ending in
///   `" PRIVATE KEY"` whose prefix names a method with an `old_priv_decode` — the legacy
///   per-algorithm decoders, which the comment at `:145-148` says engines never provide;
/// * `PARAMETERS` matches any name ending in `" PARAMETERS"` whose prefix names a method with a
///   `param_decode`;
/// * `X9.42 DH PARAMETERS` is accepted where `DH PARAMETERS` was asked for, and the reverse is
///   **not** — the arm is one-way; the three older spellings (`X509 CERTIFICATE`,
///   `NEW CERTIFICATE REQUEST`) are accepted where their modern names were asked for; a plain
///   `CERTIFICATE` is accepted as a `TRUSTED CERTIFICATE`; `CERTIFICATE` and `PKCS #7 SIGNED
///   DATA` are both accepted as `PKCS7`; and, with `OPENSSL_NO_CMS` undefined on this profile,
///   `CERTIFICATE` and `PKCS7` are both accepted as `CMS`.
///
/// The two `EVP_PKEY_asn1_find_str` arms answer through the crate's own registry, so on a crate
/// where no method is registered (`src/evp/pkey_asn1.rs`'s `STANDARD_METHODS` is empty) they
/// fall through to `0` for every prefix.
///
/// # Safety
///
/// `nm` and `name` must each be NULL or NUL-terminated.
unsafe fn check_pem(nm: *const c_char, name: *const c_char) -> c_int {
    /* Normal matching nm and name */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, name) } == 0 {
        return 1;
    }

    /* Make PEM_STRING_EVP_PKEY match any private key */
    // SAFETY: `name` is NUL-terminated per the contract.
    if unsafe { strcmp(name, PEM_STRING_EVP_PKEY) } == 0 {
        // SAFETY: `nm` is NUL-terminated per the contract.
        if unsafe { strcmp(nm, PEM_STRING_PKCS8) } == 0 {
            return 1;
        }
        // SAFETY: as above.
        if unsafe { strcmp(nm, PEM_STRING_PKCS8INF) } == 0 {
            return 1;
        }
        // SAFETY: `nm` and the static suffix are NUL-terminated.
        let slen = unsafe { ossl_pem_check_suffix(nm, c"PRIVATE KEY".as_ptr()) };
        if slen > 0 {
            /*
             * NB: ENGINE implementations won't contain a deprecated old
             * private key decode function so don't look for them.
             */
            // SAFETY: `nm` is NUL-terminated and `slen` is the prefix length just measured.
            let ameth = unsafe { EVP_PKEY_asn1_find_str(ptr::null_mut(), nm, slen) };
            // SAFETY: `ameth` is NULL or the live method the lookup just answered.
            if !ameth.is_null() && unsafe { (*ameth).old_priv_decode }.is_some() {
                return 1;
            }
        }
        return 0;
    }

    // SAFETY: `name` is NUL-terminated per the contract.
    if unsafe { strcmp(name, PEM_STRING_PARAMETERS) } == 0 {
        // SAFETY: `nm` and the static suffix are NUL-terminated.
        let slen = unsafe { ossl_pem_check_suffix(nm, c"PARAMETERS".as_ptr()) };
        if slen > 0 {
            /*
             * The authority writes `ENGINE *e;` uninitialised and reads it only on the arm
             * where the lookup answered a method. `EVP_PKEY_asn1_find_str` writes through
             * `*pe` before it can answer non-null (`src/evp/pkey_asn1.rs:1027`), so a null
             * initialiser is the same value on every path that reads it. Recorded, not a
             * divergence.
             */
            let mut e: *mut Engine = ptr::null_mut();
            // SAFETY: `nm` is NUL-terminated and `e` is this frame's own slot.
            let ameth = unsafe { EVP_PKEY_asn1_find_str(&mut e, nm, slen) };
            if !ameth.is_null() {
                // SAFETY: `ameth` is the live method the lookup just answered.
                let r: c_int = if unsafe { (*ameth).param_decode }.is_some() {
                    1
                } else {
                    0
                };
                /*
                 * `#ifndef OPENSSL_NO_ENGINE` — undefined in this profile, so the authority
                 * compiles the finish in. `ENGINE_finish` is Phase 13's and is absent; see
                 * the module's missing-callee list.
                 */
                // SAFETY: `e` is NULL or the engine the lookup answered.
                unsafe { ENGINE_finish(e) };
                return r;
            }
        }
        return 0;
    }
    /* If reading DH parameters handle X9.42 DH format too */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_DHXPARAMS) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_DHPARAMS) } == 0
    {
        return 1;
    }

    /* Permit older strings */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509_OLD) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_X509) } == 0
    {
        return 1;
    }

    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509_REQ_OLD) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_X509_REQ) } == 0
    {
        return 1;
    }

    /* Allow normal certs to be read as trusted certs */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_X509_TRUSTED) } == 0
    {
        return 1;
    }

    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509_OLD) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_X509_TRUSTED) } == 0
    {
        return 1;
    }

    /* Some CAs use PKCS#7 with CERTIFICATE headers */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_PKCS7) } == 0
    {
        return 1;
    }

    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_PKCS7_SIGNED) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_PKCS7) } == 0
    {
        return 1;
    }

    // `#ifndef OPENSSL_NO_CMS` — undefined in this profile, so both arms are compiled.
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_X509) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_CMS) } == 0
    {
        return 1;
    }
    /* Allow CMS to be read from PKCS#7 headers */
    // SAFETY: both strings are NUL-terminated per the contract.
    if unsafe { strcmp(nm, PEM_STRING_PKCS7) } == 0
        // SAFETY: as above.
        && unsafe { strcmp(name, PEM_STRING_CMS) } == 0
    {
        return 1;
    }

    return 0;
}

/// `static int pem_bytes_read_bio_flags(unsigned char **pdata, long *plen, char **pnm,
/// const char *name, BIO *bp, pem_password_cb *cb, void *u, unsigned int flags)` —
/// `crypto/pem/pem_lib.c:243-284`.
///
/// The reader's middle step, and the only place the two public spellings differ: it loops
/// because a BIO may hold several PEM blocks in sequence, discarding each one whose name
/// `check_pem` rejects, until it finds the block the caller asked for. Only then does it parse
/// the `Proc-Type`/`DEK-Info` header into a local `EVP_CIPHER_INFO` — deliberately **after** the
/// name match, so a block it is about to skip is never decrypted — and only then does it call
/// [`PEM_do_header`].
///
/// Ownership on success: `*pdata` takes the body and `*plen` its length (both flags apart);
/// `*pnm`, when non-NULL, takes the name, and the local `nm` is then *not* freed. `header` is
/// always released, decrypted or not.
///
/// # Safety
///
/// `pdata`, `plen` and `pnm` must be NULL or writable for one pointer/`long` each; `name` must
/// be NUL-terminated; `bp` must be a live BIO; `cb` is the caller's callback or `None`; `u` is
/// passed to it untouched.
unsafe fn pem_bytes_read_bio_flags(
    pdata: *mut *mut c_uchar,
    plen: *mut c_long,
    pnm: *mut *mut c_char,
    name: *const c_char,
    bp: *mut Bio,
    cb: Option<PemPasswordCb>,
    u: *mut c_void,
    flags: c_uint,
) -> c_int {
    /* EVP_CIPHER_INFO cipher; */
    let mut cipher = core::mem::MaybeUninit::<EvpCipherInfo>::uninit();
    let mut nm: *mut c_char = ptr::null_mut();
    let mut header: *mut c_char = ptr::null_mut();
    let mut data: *mut c_uchar = ptr::null_mut();
    let mut len: c_long = 0;
    let mut ret: c_int = 0;

    loop {
        /* PEM_FREE(nm, flags, 0); */
        // SAFETY: `nm` is NULL or the previous iteration's block.
        unsafe { pem_free(nm.cast(), flags, 0, FILE_PEM_LIB, 255) };
        /* PEM_FREE(header, flags, 0); */
        // SAFETY: as above.
        unsafe { pem_free(header.cast(), flags, 0, FILE_PEM_LIB, 256) };
        /* PEM_FREE(data, flags, len); */
        // SAFETY: `data` is NULL or the previous iteration's block, `len` bytes long.
        unsafe { pem_free(data.cast(), flags, len as usize, FILE_PEM_LIB, 257) };

        // SAFETY: the five out-parameters are this frame's own slots and `bp` is the caller's
        // live BIO. `PEM_read_bio_ex` NULLs all four before it parses anything.
        if unsafe { PEM_read_bio_ex(bp, &mut nm, &mut header, &mut data, &mut len, flags) } == 0 {
            // `ERR_GET_REASON(ERR_peek_error())` — `ERR_peek_error` touches only this thread's
            // queue and `ERR_GET_REASON` is the mask-and-cast macro; see the module's
            // missing-callee list for why neither is spelled the crate's way.
            if ERR_GET_REASON(ERR_peek_error()) == PEM_R_NO_START_LINE {
                // SAFETY: both arguments are NUL-terminated (`name` per the contract).
                unsafe { ERR_add_error_data(2, c"Expecting: ".as_ptr(), name) };
            }
            return 0;
        }
        // `} while (!check_pem(nm, name));` — the loop turns only when the name did **not**
        // match, so a match breaks out with `nm` still owned by this frame.
        // SAFETY: `nm` and `name` are NUL-terminated.
        if unsafe { check_pem(nm, name) } != 0 {
            break;
        }
    }
    // `if (!PEM_get_EVP_CIPHER_INFO(header, &cipher)) goto err;`
    // SAFETY: `header` is the header `PEM_read_bio_ex` just handed over and `cipher` is this
    // frame's own slot.
    if unsafe { PEM_get_EVP_CIPHER_INFO(header, cipher.as_mut_ptr()) } == 0 {
        // SAFETY: the five arguments are this frame's own blocks and slots.
        return unsafe { pem_bytes_read_bio_err(nm, header, data, len, flags, ret, pnm) };
    }
    // `if (!PEM_do_header(&cipher, data, &len, cb, u)) goto err;`
    // SAFETY: `cipher` was filled by the call above, and `data`/`len` are the body it produced.
    if unsafe { PEM_do_header(cipher.as_mut_ptr(), data, &mut len, cb, u) } == 0 {
        // SAFETY: as above.
        return unsafe { pem_bytes_read_bio_err(nm, header, data, len, flags, ret, pnm) };
    }

    // SAFETY: the two out-parameters are the caller's own slots.
    unsafe {
        *pdata = data;
        *plen = len;
    }

    if !pnm.is_null() {
        // SAFETY: `pnm` is non-NULL and so, per the contract, is the caller's writable slot.
        unsafe { *pnm = nm };
    }

    ret = 1;

    // `err:`
    // SAFETY: as above.
    unsafe { pem_bytes_read_bio_err(nm, header, data, len, flags, ret, pnm) }
}

/// `err:` in [`pem_bytes_read_bio_flags`] — `crypto/pem/pem_lib.c:277-283`.
///
/// One place for the label's three frees, because two `goto err`s and the fall-through all land
/// here. Which of the three run depends on `ret` and on whether the caller wanted the name:
///
/// | free | condition |
/// |---|---|
/// | `nm` | `ret == 0` **or** `pnm == NULL` |
/// | `header` | always |
/// | `data` | `ret == 0` |
///
/// `PEM_FREE` carries `OPENSSL_LINE`, so each free records the line it is written on — the three
/// here are 279, 280 and 282.
///
/// # Safety
///
/// `nm`, `header` and `data` must be NULL or blocks this frame owns (from
/// `PEM_read_bio_ex`), `len` the length `data` was allocated with, and `pnm` the caller's slot.
unsafe fn pem_bytes_read_bio_err(
    nm: *mut c_char,
    header: *mut c_char,
    data: *mut c_uchar,
    len: c_long,
    flags: c_uint,
    ret: c_int,
    pnm: *mut *mut c_char,
) -> c_int {
    if ret == 0 || pnm.is_null() {
        /* PEM_FREE(nm, flags, 0); */
        // SAFETY: `nm` is this frame's block or NULL.
        unsafe { pem_free(nm.cast(), flags, 0, FILE_PEM_LIB, 279) };
    }
    /* PEM_FREE(header, flags, 0); */
    // SAFETY: `header` is this frame's block or NULL.
    unsafe { pem_free(header.cast(), flags, 0, FILE_PEM_LIB, 280) };
    if ret == 0 {
        /* PEM_FREE(data, flags, len); */
        // SAFETY: `data` is this frame's block, `len` bytes long, or NULL.
        unsafe { pem_free(data.cast(), flags, len as usize, FILE_PEM_LIB, 282) };
    }
    ret
}

/// `int PEM_do_header(EVP_CIPHER_INFO *cipher, unsigned char *data, long *len,
/// pem_password_cb *callback, void *u)` — `crypto/pem/pem_lib.c:445-504`.
///
/// Decrypts the body of an encrypted PEM block in place, and is the hinge every
/// `PEM_read_bio_*` reaches: `PEM_bytes_read_bio`, `pem_bytes_read_bio_flags`,
/// `PEM_ASN1_read_bio` — and through it `PEM_ASN1_read`.
///
/// The order of the guards is observable, and the three early returns are **not**
/// interchangeable:
///
/// 1. a length above `INT_MAX` is refused (`PEM_R_HEADER_TOO_LONG`) **before** the cipher is
///    looked at, so a null cipher cannot rescue it; `#if LONG_MAX > INT_MAX` holds for `c_long`
///    on this target, so the arm is compiled;
/// 2. a null `cipher->cipher` answers success immediately — the block was not encrypted — and
///    **no password is read**, which is the arm a caller can drive with no callback at all;
/// 3. a negative `keylen` is `PEM_R_BAD_PASSWORD_READ`.
///
/// On the three failure returns above, `buf` and `key` are **not** cleansed — the authority
/// returns from inside the body. Only the path that reaches the end clears them, after
/// `EVP_CIPHER_CTX_free`. `EVP_DecryptUpdate`/`_Final_ex` share one `int` length variable, and
/// `*len` is updated after the update (so a final-block failure still reports the plaintext
/// decrypted so far) and again after the final.
///
/// # Safety
///
/// `cipher` must be a live, initialised `EVP_CIPHER_INFO`; `data` must be writable for
/// `*plen` bytes plus one cipher block; `plen` must be a writable `long`; `callback`, when
/// `Some`, must be a live callback and `u` is passed to it untouched.
#[no_mangle]
pub unsafe extern "C" fn PEM_do_header(
    cipher: *mut EvpCipherInfo,
    data: *mut c_uchar,
    plen: *mut c_long,
    callback: Option<PemPasswordCb>,
    u: *mut c_void,
) -> c_int {
    let mut ok: c_int;
    let mut keylen: c_int;
    // `long len = *plen; int ilen = (int)len;` — the narrowing is the authority's and happens
    // before the check below.
    // SAFETY: `plen` is the caller's writable slot.
    let len: c_long = unsafe { *plen };
    let mut ilen: c_int = len as c_int;
    // `EVP_CIPHER_CTX *ctx;` — created below, so it is not declared yet.
    let mut key = [0u8; EVP_MAX_KEY_LENGTH as usize];
    let mut buf = [0 as c_char; PEM_BUFSIZE as usize];

    /* Check that we did not truncate the length */
    if len > c_int::MAX as c_long {
        // SAFETY: a compile-time-constant site.
        unsafe { raise_site(&err_sites::PEM_LIB_459) };
        return 0;
    }

    // SAFETY: `cipher` is the caller's live structure.
    if unsafe { (*cipher).cipher }.is_null() {
        return 1;
    }
    keylen = match callback {
        // The authority's `PEM_def_callback(buf, PEM_BUFSIZE, 0, u)`; see the module's
        // missing-callee list.
        None => unsafe {
            crate::pem::pem_lib::PEM_def_callback(buf.as_mut_ptr(), PEM_BUFSIZE, 0, u)
        },
        // SAFETY: `buf` is `PEM_BUFSIZE` writable bytes and `u` is the caller's.
        Some(cb) => unsafe { cb(buf.as_mut_ptr(), PEM_BUFSIZE, 0, u) },
    };
    if keylen < 0 {
        // SAFETY: a compile-time-constant site.
        unsafe { raise_site(&err_sites::PEM_LIB_471) };
        return 0;
    }
    /* #ifdef CHARSET_EBCDIC is not defined on this profile, so the ebcdic2ascii call is not
     * compiled. */

    // SAFETY: `cipher->cipher` is live, `EVP_md5()` is the crate's static legacy object,
    // `cipher->iv` is the salt the header carried, and `buf`/`key` are this frame's buffers.
    if unsafe {
        EVP_BytesToKey(
            (*cipher).cipher,
            EVP_md5(),
            (*cipher).iv.as_ptr(),
            buf.as_ptr().cast::<c_uchar>(),
            keylen,
            1,
            key.as_mut_ptr(),
            ptr::null_mut(),
        )
    } == 0
    {
        return 0;
    }

    // SAFETY: allocates a fresh context and answers NULL on failure, which is checked below.
    let ctx = EVP_CIPHER_CTX_new();
    if ctx.is_null() {
        return 0;
    }

    // SAFETY: `ctx` is live, `cipher->cipher` is live, and `key`/`cipher->iv` are the frame's
    // own buffers of the lengths the cipher was asked for.
    ok = unsafe {
        EVP_DecryptInit_ex(
            ctx,
            (*cipher).cipher,
            ptr::null_mut(),
            key.as_ptr(),
            (*cipher).iv.as_ptr(),
        )
    };
    if ok != 0 {
        // SAFETY: `data` is the caller's `ilen` bytes with room for a final block, and `ilen`
        // is this frame's own length.
        ok = unsafe { EVP_DecryptUpdate(ctx, data, &mut ilen, data, ilen) };
    }
    if ok != 0 {
        /* Squirrel away the length of data decrypted so far. */
        // SAFETY: `plen` is the caller's writable slot.
        unsafe { *plen = ilen as c_long };
        // SAFETY: the caller's contract leaves room for one cipher block past `ilen` bytes.
        ok = unsafe { EVP_DecryptFinal_ex(ctx, data.add(ilen as usize), &mut ilen) };
    }
    if ok != 0 {
        // SAFETY: `plen` is the caller's writable slot.
        unsafe { *plen += ilen as c_long };
    } else {
        // SAFETY: a compile-time-constant site.
        unsafe { raise_site(&err_sites::PEM_LIB_498) };
    }

    // SAFETY: `ctx` is this frame's own context, and `buf`/`key` are this frame's own buffers
    // of exactly these lengths.
    unsafe {
        EVP_CIPHER_CTX_free(ctx);
        OPENSSL_cleanse(buf.as_mut_ptr().cast(), PEM_BUFSIZE as usize);
        OPENSSL_cleanse(key.as_mut_ptr().cast(), EVP_MAX_KEY_LENGTH as usize);
    }
    ok
}

/// `void *PEM_ASN1_read(d2i_of_void *d2i, const char *name, FILE *fp, void **x,
/// pem_password_cb *cb, void *u)` — `crypto/pem/pem_lib.c:111-125`.
///
/// `PEM_ASN1_read_bio` with a `BIO_s_file()` pushed onto the caller's `FILE *` and
/// `BIO_NOCLOSE` set, so the stream is not closed under the caller. Compiled because this
/// profile leaves `OPENSSL_NO_STDIO` undefined. The BIO is released on **both** paths — a null
/// `BIO_new` raises `ERR_R_BUF_LIB` at `:118` and answers `0`.
///
/// # Safety
///
/// As [`PEM_ASN1_read_bio`]; `fp` must be an open, readable `FILE *` that outlives the call.
#[no_mangle]
pub unsafe extern "C" fn PEM_ASN1_read(
    d2i: D2iOfVoid,
    name: *const c_char,
    fp: *mut c_void,
    x: *mut *mut c_void,
    cb: Option<PemPasswordCb>,
    u: *mut c_void,
) -> *mut c_void {
    /* if ((b = BIO_new(BIO_s_file())) == NULL) { */
    // SAFETY: `BIO_s_file()` is a static method table.
    let b = unsafe { BIO_new(BIO_s_file()) };
    if b.is_null() {
        // SAFETY: a compile-time-constant site.
        unsafe { raise_site(&err_sites::PEM_LIB_118) };
        return ptr::null_mut();
    }
    /* BIO_set_fp(b, fp, BIO_NOCLOSE); */
    // The macro at `include/openssl/bio.h:568` is
    // `BIO_ctrl(b, BIO_C_SET_FILE_PTR, c, (char *)(fp))`.
    // SAFETY: `b` is this frame's live file BIO and `fp` is the caller's stream.
    unsafe { BIO_ctrl(b, BIO_C_SET_FILE_PTR, BIO_NOCLOSE, fp) };
    /* ret = PEM_ASN1_read_bio(d2i, name, b, x, cb, u); */
    // SAFETY: `b` is live and the remaining arguments are the caller's.
    let ret = unsafe { PEM_ASN1_read_bio(d2i, name, b, x, cb, u) };
    /* BIO_free(b); */
    // SAFETY: `b` is this frame's own file BIO.
    unsafe { BIO_free(b) };
    ret
}

/// `void *PEM_ASN1_read_bio(d2i_of_void *d2i, const char *name, BIO *bp, void **x,
/// pem_password_cb *cb, void *u)` — `crypto/pem/pem_oth.c:20-36`.
///
/// The generic `PEM_read_<type>_bio` body: read one `name`-typed block with
/// [`PEM_bytes_read_bio`] (no name wanted, so the third argument is NULL), hand the decoded
/// bytes to the caller's `d2i`, and release them.
///
/// Two details are observable and are kept: the decode's cursor is a **local** `const unsigned
/// char *p` initialised from the buffer, so the caller's `x` is the only thing `d2i` writes
/// through; and the buffer is released on **both** the successful and the failed decode — a
/// caller cannot tell "bad DER" from "no block" apart, because both answer NULL.
///
/// # Safety
///
/// `d2i` must be a live decoder matching `x`'s type; `name` must be NUL-terminated; `bp` must be
/// a live BIO; `x` must be a live slot for the decoded object.
#[no_mangle]
pub unsafe extern "C" fn PEM_ASN1_read_bio(
    d2i: D2iOfVoid,
    name: *const c_char,
    bp: *mut Bio,
    x: *mut *mut c_void,
    cb: Option<PemPasswordCb>,
    u: *mut c_void,
) -> *mut c_void {
    // `const unsigned char *p = NULL;` — the cursor is set from `data` below.
    let mut data: *mut c_uchar = ptr::null_mut();
    let mut len: c_long = 0;

    // SAFETY: `data` and `len` are this frame's own slots, `bp` is the caller's live BIO, the
    // `pnm` argument is the authority's own NULL, and the rest are the caller's.
    if unsafe {
        PEM_bytes_read_bio(&mut data, &mut len, ptr::null_mut(), name, bp, cb, u)
    } == 0
    {
        return ptr::null_mut();
    }
    /* p = data; */
    let mut p: *const c_uchar = data;
    /* ret = d2i(x, &p, len); */
    // SAFETY: `data` holds `len` bytes — what `PEM_bytes_read_bio` just handed over — and `d2i`
    // is the caller's decoder.
    let ret = unsafe { d2i(x, &mut p, len) };
    if ret.is_null() {
        // SAFETY: a compile-time-constant site.
        unsafe { raise_site(&err_sites::PEM_OTH_33) };
    }
    /* OPENSSL_free(data); */
    // The macro over `CRYPTO_free(data, OPENSSL_FILE, OPENSSL_LINE)` at `pem_oth.c:34`.
    // SAFETY: `data` is the block `PEM_bytes_read_bio` just handed over and is not owned
    // elsewhere.
    unsafe { CRYPTO_free(data.cast(), FILE_PEM_OTH, LINE_PEM_OTH_FREE) };
    ret
}

// =============================================================================================
// Tests. Every case below is one of the three kinds the brief admits: a refusal arm, an
// argument-validation arm, or a path that needs neither randomness nor a provider fetch. The
// arms that do decrypt are not tested here, because a cipher fetch is a provider round trip and
// the court for that is `RT-EVP-PEM`'s, not this module's.
// =============================================================================================

#[cfg(test)]
mod tests {
    use super::*;
    use crate::runtime::bio::bss_mem::BIO_s_mem;
    use crate::runtime::err::{err_sites, ERR_clear_error, ERR_peek_error};
    use core::ffi::c_ulong;
    use core::sync::atomic::{AtomicUsize, Ordering};

    /// The packed code `ERR_peek_error` reports for a recorded site:
    /// `(lib & 0xff) << 23 | (reason & 0x7fffff)`, read from the generated table rather than
    /// typed, so a wrong expectation here cannot disagree with the authority's coordinates.
    fn packed(site: &err_sites::ErrSite) -> c_ulong {
        (((site.lib as c_ulong) & 0xff) << 23) | ((site.reason as c_ulong) & 0x7f_ffff)
    }

    /// An `EVP_CIPHER_INFO` as `PEM_get_EVP_CIPHER_INFO` leaves one for a block with no header.
    fn no_cipher() -> EvpCipherInfo {
        EvpCipherInfo {
            cipher: ptr::null(),
            iv: [0; 16],
        }
    }

    /// `check_pem` answers for each alias the authority's chain declares. These are pure
    /// `strcmp` arms: no allocation, no registry read, no provider.
    #[test]
    fn check_pem_matches_the_authority_aliases() {
        let names: [(&core::ffi::CStr, &core::ffi::CStr); 11] = [
            (c"X509 CERTIFICATE", c"CERTIFICATE"),
            (c"NEW CERTIFICATE REQUEST", c"CERTIFICATE REQUEST"),
            (c"CERTIFICATE", c"TRUSTED CERTIFICATE"),
            (c"X509 CERTIFICATE", c"TRUSTED CERTIFICATE"),
            (c"CERTIFICATE", c"PKCS7"),
            (c"PKCS #7 SIGNED DATA", c"PKCS7"),
            (c"X9.42 DH PARAMETERS", c"DH PARAMETERS"),
            (c"ENCRYPTED PRIVATE KEY", c"ANY PRIVATE KEY"),
            (c"PRIVATE KEY", c"ANY PRIVATE KEY"),
            // The `#ifndef OPENSSL_NO_CMS` pair, compiled because this profile configures CMS in.
            (c"CERTIFICATE", c"CMS"),
            (c"PKCS7", c"CMS"),
        ];
        for (nm, name) in names {
            // SAFETY: both are NUL-terminated literals.
            assert_eq!(
                unsafe { check_pem(nm.as_ptr(), name.as_ptr()) },
                1,
                "nm={nm:?} name={name:?}"
            );
        }
    }

    /// The refusals, including the two the direction of an arm decides: the X9.42 spelling is
    /// accepted where `DH PARAMETERS` was asked for and the reverse is not, and `ANY PRIVATE
    /// KEY` is a header name, not a request.
    #[test]
    fn check_pem_refuses_what_no_alias_covers() {
        let names: [(&core::ffi::CStr, &core::ffi::CStr); 4] = [
            (c"CERTIFICATE", c"RSA PRIVATE KEY"),
            (c"PUBLIC KEY", c"RSA PRIVATE KEY"),
            (c"DH PARAMETERS", c"X9.42 DH PARAMETERS"),
            (c"ANY PRIVATE KEY", c"CERTIFICATE"),
        ];
        for (nm, name) in names {
            // SAFETY: both are NUL-terminated literals.
            assert_eq!(
                unsafe { check_pem(nm.as_ptr(), name.as_ptr()) },
                0,
                "nm={nm:?} name={name:?}"
            );
        }
    }

    /// A block with no `DEK-Info` has a null `cipher->cipher`, and the authority answers success
    /// **before** it reads the callback — so a null callback is legal here and the body and
    /// length come back untouched. No cipher is fetched, so no provider is reached.
    #[test]
    fn a_null_cipher_answers_success_without_reading_a_password() {
        ERR_clear_error();
        let mut cipher = no_cipher();
        let mut body = [0u8; 4];
        let mut len: c_long = 4;

        // SAFETY: `cipher`, `body` and `len` are this test's own objects, and the callback is
        // null on purpose: the arm returns before it could be called.
        let r = unsafe {
            PEM_do_header(
                &mut cipher,
                body.as_mut_ptr(),
                &mut len,
                None,
                ptr::null_mut(),
            )
        };
        assert_eq!(r, 1);
        assert_eq!(len, 4, "the length is left alone");
        assert_eq!(body, [0u8; 4], "and so is the body");
        assert_eq!(ERR_peek_error(), 0, "and nothing is raised");
    }

    /// The length guard is **before** the null-cipher guard, so a length above `INT_MAX` is
    /// refused even for a block that would otherwise answer success. The null cipher is what
    /// makes the ordering visible.
    #[test]
    fn a_length_above_int_max_is_refused_before_the_cipher_is_read() {
        ERR_clear_error();
        let mut cipher = no_cipher();
        let mut body = [0u8; 1];
        let mut len: c_long = c_int::MAX as c_long + 1;

        // SAFETY: this test's own objects.
        let r = unsafe {
            PEM_do_header(
                &mut cipher,
                body.as_mut_ptr(),
                &mut len,
                None,
                ptr::null_mut(),
            )
        };
        assert_eq!(r, 0);
        assert_eq!(ERR_peek_error(), packed(&err_sites::PEM_LIB_459));
    }

    /// An empty BIO carries no `-----BEGIN `, so the reader refuses with `get_name`'s
    /// `PEM_R_NO_START_LINE` and a caller's three out-parameters are left null/zero — the
    /// authority's early `return 0`, not its `err:` tail.
    #[test]
    fn an_empty_bio_is_refused_with_the_no_start_line_reason() {
        ERR_clear_error();
        // SAFETY: `BIO_s_mem()` is a static method table.
        let bp = unsafe { BIO_new(BIO_s_mem()) };
        assert!(!bp.is_null(), "the memory BIO is allocated");
        let mut data: *mut c_uchar = ptr::null_mut();
        let mut len: c_long = 0;
        let mut nm: *mut c_char = ptr::null_mut();

        // SAFETY: `bp` is live and the three out-parameters are this test's own slots.
        let r = unsafe {
            PEM_bytes_read_bio(
                &mut data,
                &mut len,
                &mut nm,
                c"CERTIFICATE".as_ptr(),
                bp,
                None,
                ptr::null_mut(),
            )
        };
        assert_eq!(r, 0);
        assert!(data.is_null(), "no body is published");
        assert!(nm.is_null(), "and no name");
        assert_eq!(len, 0);
        assert_eq!(ERR_peek_error(), packed(&err_sites::PEM_LIB_794));

        // SAFETY: `bp` is this test's BIO.
        unsafe { BIO_free(bp) };
    }

    /// A decoder that must never be reached. `PEM_ASN1_read_bio` calls it only after
    /// `PEM_bytes_read_bio` has answered success, so an empty BIO proves the order.
    static DECODER_CALLS: AtomicUsize = AtomicUsize::new(0);

    /// The counting decoder itself. Its return value is never used by the test.
    ///
    /// # Safety
    /// The signature is `d2i_of_void`'s; the test never allows it to be called.
    unsafe extern "C" fn counting_d2i(
        _x: *mut *mut c_void,
        _p: *mut *const c_uchar,
        _len: c_long,
    ) -> *mut c_void {
        DECODER_CALLS.fetch_add(1, Ordering::SeqCst);
        ptr::null_mut()
    }

    /// The refusal comes *before* the decode: no block, no `d2i` call.
    #[test]
    fn asn1_read_bio_returns_null_without_calling_the_decoder() {
        ERR_clear_error();
        DECODER_CALLS.store(0, Ordering::SeqCst);
        // SAFETY: `BIO_s_mem()` is a static method table.
        let bp = unsafe { BIO_new(BIO_s_mem()) };
        assert!(!bp.is_null());
        let mut x: *mut c_void = ptr::null_mut();

        // SAFETY: `bp` is live, `x` is this test's own slot, and `counting_d2i` is a live
        // `d2i_of_void` that the call must not reach.
        let r = unsafe {
            PEM_ASN1_read_bio(
                counting_d2i,
                c"CERTIFICATE".as_ptr(),
                bp,
                &mut x,
                None,
                ptr::null_mut(),
            )
        };
        assert!(r.is_null());
        assert!(x.is_null(), "the caller's slot is untouched");
        assert_eq!(
            DECODER_CALLS.load(Ordering::SeqCst),
            0,
            "the decoder is not reached"
        );

        // SAFETY: `bp` is this test's BIO.
        unsafe { BIO_free(bp) };
    }
}
