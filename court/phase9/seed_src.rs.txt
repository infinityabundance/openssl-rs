//! Phase 9 staging — the providers' seed-source / entropy units.
//!
//! **This is a `.rs.txt` hand-integration staging file, not a compiled module.** It is
//! destined for `src/provider/rand.rs` (or equivalent), and it is written to the crate's
//! conventions so that integration is a move rather than a rewrite. It deliberately does
//! *not* compile: symbols the crate does not yet have are called by name and listed in the
//! `MISSING` inventory at the bottom. Nothing here defines one of them.
//!
//! ## Sources read, in full
//!
//! * `providers/implementations/rands/seed_src.c` — generated from `seed_src.c.in`. The
//!   `{- produce_param_decoder(...) -}` template directive expands in the build tree, and the
//!   generated file (`forensics/authorities/build/openssl-3.6.4-production/.../seed_src.c`,
//!   327 lines) is what is transcribed here, because the machine-generated decoder's line
//!   numbers and `__FILE__` are what the build record records.
//! * `providers/implementations/rands/test_rng.c` — generated from `test_rng.c.in`
//!   (571 lines).
//! * `providers/implementations/rands/fips_crng_test.c` — generated from
//!   `fips_crng_test.c.in` (526 lines). **This is the continuous RNG test in 3.6.4.**
//! * `crypto/rand/rand_uniform.c` (109 lines), which `test_rng.c`'s siblings reference and
//!   which the RAND stratum needs; transcribed here so the unit is complete.
//! * `providers/baseprov.c` — **the RAND row only**, reproduced as a Rust row description below.
//! * `providers/implementations/include/prov/names.h` (the `PROV_NAMES_*` RAND block, lines
//!   325-335) for the exact `algorithm_names` strings.
//! * `providers/defltprov.c` and `providers/fips/fipsprov.c` were consulted *only* to locate
//!   the rows named in the task; neither is transcribed.
//!
//! ## Discrepancies between the task's description and the admitted 3.6.4 authority
//!
//! These are recorded, not papered over. **The names the task gives for two of the three units
//! do not exist anywhere in the 3.6.4 authority tree**:
//!
//! 1. **There is no `crngt.c` and no `ossl_crngt_init` / `ossl_crngt_get_entropy` /
//!    `ossl_crngt_cleanup` / `CRNGT_BLOCK_SIZE` / `crypto_get_entropy` in 3.6.4.** Those are
//!    the OpenSSL 3.0-era `providers/implementations/rands/crngt.c` symbols. In 3.6.4 the
//!    continuous test is `fips_crng_test.c`, whose whole surface is `crng_test_new`/`_free`/
//!    `_instantiate`/`_uninstantiate`/`_generate`/`_reseed`, the SP 800-90B `RCT_test` and
//!    `APT_test`, and `ossl_crng_test_functions`. A dead `extern const OSSL_DISPATCH
//!    crngt_functions[];` survives in `prov/implementations.h:316` with no definition — that
//!    declaration is the only trace of the old name. `fips_crng_test.c` is transcribed in full.
//! 2. **There is no `seed_src_is_entropy`, no `seed_src_copy`, and no
//!    `ossl_prov_seed_src_get_ctx_params` / `_set_ctx_params` / `_gettable` / `_settable` in
//!    3.6.4's `seed_src.c`.** The 3.6.4 unit has no `set_ctx_params` and no `settable_ctx_params`
//!    at all (SEED-SRC publishes no settable keys), and its getter is the static
//!    `seed_src_get_ctx_params`. The whole presented unit is transcribed instead.
//! 3. **`providers/baseprov.c` publishes no TEST-RAND row.** Its `base_rands[]` carries
//!    `PROV_NAMES_SEED_SRC` alone in this build (the `JITTER` row is `#ifndef
//!    OPENSSL_NO_JITTER`-guarded and `OPENSSL_NO_JITTER` **is defined** in
//!    `configdata.pm:215`). TEST-RAND lives in the *default* provider at
//!    `defltprov.c:412`. Both facts are reproduced below.
//! 4. **`PROV_NAMES_SEED_SRC` is `"SEED-SRC"` and `PROV_NAMES_TEST_RAND` is `"TEST-RAND"`,
//!    with no alias and no OID**, so "every alias/OID it declares" is the empty set. (The task
//!    calls the latter `PROV_NAMES_TEST_RNG`; the authority's macro is `PROV_NAMES_TEST_RAND`.)
//! 5. `test_rng.c` has **no** `ossl_test_rng_get_size` and **no** `ossl_test_rng_set_state`
//!    helper — the task's "if present" resolves to absent.
//!
//! ## Build facts this transcription depends on
//!
//! * `FIPS_MODULE` is undefined for the default/base providers, so every `# if
//!   defined(FIPS_MODULE)` arm is absent: `test_rng`'s and `fips_crng_test`'s `get_ctx_params`
//!   lose their `fips-indicator` key, and `fips_crng_test`'s `get_ctx_params` loses its `ind`
//!   field. (Confirmed by the generated files, which keep the `#ifdef`/`#endif` and by
//!   `configdata.pm`'s disabled list.) The arms are preserved below in comments so integration
//!   can see what is being omitted.
//! * `OPENSSL_NO_JITTER` and `OPENSSL_NO_FIPS_SECURITYCHECKS` are in that disabled list, so the
//!   `JITTER` rows are absent and `prov/securitycheck.h`'s macros are no-ops.
//!
//! ## The three `FILE` strings
//!
//! `seed_src.c`, `test_rng.c` and `fips_crng_test.c` are all generated from `.c.in` templates,
//! so `__FILE__` is the **build-relative** path with no `../../src/openssl-3.6.4/` prefix —

/// `typedef struct { void *provctx; int state; } PROV_SEED_SRC` — `seed_src.c:46-49`.
#[repr(C)]
pub(crate) struct ProvSeedSrc {
    /// `void *provctx` — the creating provider's context; unused by every arm of this unit.

    pub state: c_int,

/// `static void *seed_src_new(void *provctx, void *parent, const OSSL_DISPATCH
/// *parent_dispatch)` — `seed_src.c:51-68`.
///
/// **A seed source must not have a parent**, and the refusal is a *raised*
/// `PROV_R_SEED_SOURCES_MUST_NOT_HAVE_A_PARENT` before any allocation. The third argument is
/// part of the `newctx` contract and is ignored here.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn seed_src_new(
    provctx: *mut c_void,
    parent: *mut c_void,
    _parent_dispatch: *const OsslDispatch,
) -> *mut c_void {
    // SAFETY: the caller's contract; `parent` is NULL or the parent context the core supplied.
    unsafe {
        if !parent.is_null() {
            // MISSING site: `err_sites::PROV_SEED_SRC_55` (`seed_src.c:55`,
            // ERR_LIB_PROV / PROV_R_SEED_SOURCES_MUST_NOT_HAVE_A_PARENT).
            raise_site(&err_sites::PROV_SEED_SRC_55);
            return ptr::null_mut();
        }

        let s = CRYPTO_zalloc(
            core::mem::size_of::<ProvSeedSrc>(),
            FILE_SEED_SRC,
            LINE_SEED_SRC_ZALLOC,
        )
        .cast::<ProvSeedSrc>();
        if s.is_null() {
            return ptr::null_mut();
        }

        (*s).provctx = provctx;
        (*s).state = EVP_RAND_STATE_UNINITIALISED;
        s.cast()
    }
}

/// `static int seed_src_generate(void *vseed, unsigned char *out, size_t outlen,
/// unsigned int strength, ossl_unused int prediction_resistance,
/// const unsigned char *adin, size_t adin_len)` — `seed_src.c:94-130`.
///
/// **The seed source's generate is a poll of the system entropy sources**, not a DRBG draw.
/// It is the one arm that reaches the `RAND_POOL` machinery: `ossl_rand_pool_new(strength, 1,
/// outlen, outlen)` asks for `outlen` bytes at `strength`, secure (`1`); `ossl_pool_acquire_entropy`
/// polls; a positive answer mixes `adin` in and copies the pool out. A *weak* pool
/// (`entropy_available == 0`) is **not** an error here — the function returns 0 without raising,
/// which is the shape the caller (`seed_get_seed`) then turns into a raise.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn seed_src_generate(
    vseed: *mut c_void,
    out: *mut c_uchar,
    outlen: usize,
    strength: c_uint,
    _prediction_resistance: c_int,
    adin: *const c_uchar,
    adin_len: usize,
) -> c_int {
    // SAFETY: `vseed` is a live `ProvSeedSrc` per the contract; `out` is writable for `outlen`.
    unsafe {
        let s = vseed.cast::<ProvSeedSrc>();

        if (*s).state != EVP_RAND_STATE_READY {
            // MISSING site: `err_sites::PROV_SEED_SRC_103` (`seed_src.c:103`), a
            // **dynamic-reason** site: PROV_R_IN_ERROR_STATE when `state == ERROR`, else
            // PROV_R_NOT_INSTANTIATED.
            let reason = if (*s).state == EVP_RAND_STATE_ERROR {
                crate::runtime::err_reasons::PROV_R_IN_ERROR_STATE
            } else {
                crate::runtime::err_reasons::PROV_R_NOT_INSTANTIATED
            };
            raise_site_dynamic(&err_sites::PROV_SEED_SRC_103, reason);
            return 0;
        }

        // MISSING: `ossl_rand_pool_new`, the `RAND_POOL` type and its accessors
        // (`crate::rand::rand_pool`).
        let pool: *mut RAND_POOL = ossl_rand_pool_new(strength as c_int, 1, outlen, outlen);
        if pool.is_null() {
            // MISSING site: `err_sites::PROV_SEED_SRC_111` (`seed_src.c:111`,
            // ERR_LIB_PROV / ERR_R_RAND_LIB).
            raise_site(&err_sites::PROV_SEED_SRC_111);
            return 0;
        }

        /* Get entropy by polling system entropy sources. */
        // MISSING: `ossl_pool_acquire_entropy`.
        let entropy_available = ossl_pool_acquire_entropy(pool);

        if entropy_available > 0 {
            // MISSING: `ossl_rand_pool_adin_mix_in`, `ossl_rand_pool_free`.
            if ossl_rand_pool_adin_mix_in(pool, adin, adin_len) == 0 {
                ossl_rand_pool_free(pool);
                return 0;
            }
            // MISSING: `ossl_rand_pool_buffer`, `ossl_rand_pool_length`.
            ptr::copy_nonoverlapping(
                ossl_rand_pool_buffer(pool),
                out,
                ossl_rand_pool_length(pool),
            );
        }

        // MISSING: `ossl_rand_pool_free`.
        ossl_rand_pool_free(pool);
        c_int::from(entropy_available > 0)
    }
}

/// The three keys `seed_src_get_ctx_params_decoder` locates, each with the site of its own
/// repeated-parameter raise (`seed_src.c:183`, `:202`, `:213`).
///
/// MISSING sites: `err_sites::PROV_SEED_SRC_183`, `PROV_SEED_SRC_202`, `PROV_SEED_SRC_213`.
const SEED_SRC_GET_CTX_PARAMS_DECODER_KEYS: [(&err_sites::ErrSite, *const c_char); 3] = [
    (&err_sites::PROV_SEED_SRC_183, OSSL_RAND_PARAM_MAX_REQUEST),
    (&err_sites::PROV_SEED_SRC_202, OSSL_RAND_PARAM_STATE),
    (&err_sites::PROV_SEED_SRC_213, OSSL_RAND_PARAM_STRENGTH),
];

/// `static int seed_src_get_ctx_params(void *vseed, OSSL_PARAM params[])` —
/// `seed_src.c:158-175`.
///
/// The answers are the authority's own constants: `strength` is **1024** and `max_request` is
/// **128**, neither of them derived from the context. Only `state` reads the context.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn seed_src_get_ctx_params(
    vseed: *mut c_void,
    params: *mut OsslParam,
) -> c_int {
    // SAFETY: the caller's contract; `vseed` is live and `params` is NULL or a key-terminated
    // array.
    unsafe {
        if vseed.is_null() {
            return 0;
        }
        if let Some(site) = repeated_param_site(params, &SEED_SRC_GET_CTX_PARAMS_DECODER_KEYS) {
            return fail_at(site);
        }

        let s = vseed.cast::<ProvSeedSrc>();

        let p = OSSL_PARAM_locate(params, OSSL_RAND_PARAM_STATE);
        if !p.is_null() && OSSL_PARAM_set_int(p, (*s).state) == 0 {
            return 0;
        }

        let p = OSSL_PARAM_locate(params, OSSL_RAND_PARAM_STRENGTH);
        if !p.is_null() && OSSL_PARAM_set_uint(p, 1024) == 0 {
            return 0;
        }

        let p = OSSL_PARAM_locate(params, OSSL_RAND_PARAM_MAX_REQUEST);
        if !p.is_null() && OSSL_PARAM_set_size_t(p, 128) == 0 {
            return 0;
        }
        1
    }
}

/// `static const OSSL_PARAM *seed_src_gettable_ctx_params(ossl_unused void *vseed,

///
/// The seed-source *seed* path. It differs from `seed_src_generate` in one contract detail
/// that matters: the pool is sized `min_len..max_len`, and **both** a weak pool and a failed
/// `adin` mix-in raise `PROV_R_ENTROPY_SOURCE_STRENGTH_TOO_WEAK` before the pool is freed. The
/// answer is `ossl_rand_pool_length(pool)`, and the buffer is **detached** so the caller owns
/// it — the reason `seed_clear_seed` below is `OPENSSL_secure_clear_free`.
///
/// # Safety

unsafe extern "C" fn seed_get_seed(
    _vseed: *mut c_void,
    pout: *mut *mut c_uchar,

    max_len: usize,
    _prediction_resistance: c_int,
    adin: *const c_uchar,
    adin_len: usize,

        if pool.is_null() {

            // ERR_LIB_PROV / ERR_R_RAND_LIB).
            raise_site(&err_sites::PROV_SEED_SRC_199);
            return 0;
        }

        /* Get entropy by polling system entropy sources. */
        // MISSING: `ossl_pool_acquire_entropy`.

        // MISSING: `ossl_rand_pool_adin_mix_in`, `ossl_rand_pool_length`,

/// `static unsigned char gen_byte(PROV_TEST_RNG *t)` — `test_rng.c:112-130`.
///
/// The 32-bit xorshift of Marsaglia (JSS 8(14), 2008): `n ^= n << 13; n ^= n >> 17;
/// n ^= n << 5`, keeping the low byte. Every shift here is a *wrapping* shift on a `u32`
/// and must not be allowed to trip `overflow-checks`; the XOR-shift itself is bit-exact.
fn gen_byte(t: *mut ProvTestRng) -> c_uchar {
    // SAFETY: `t` is a live `ProvTestRng` per the caller's contract.
    let mut n = unsafe { (*t).seed };
    n ^= n << 13;
    n ^= n >> 17;
    n ^= n << 5;
    // SAFETY: as above.
    unsafe { (*t).seed = n };
    (n & 0xff) as c_uchar
}

/// `static int test_rng_generate(void *vtest, unsigned char *out, size_t outlen,
/// unsigned int strength, int prediction_resistance, const unsigned char *adin,
/// size_t adin_len)` — `test_rng.c:132-152`.
///
/// Two modes. With `generate` set, `outlen` xorshift bytes. Otherwise a **prefix of the stored
/// entropy**, refused when `entropy_len - entropy_pos < outlen` — a `size_t` subtraction that
/// in C is unsigned wraparound and therefore `wrapping_sub` here, because `entropy_pos` can
/// exceed `entropy_len` after a caller rewinds or re-instantiate. The refusal raises nothing.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn test_rng_generate(
    vtest: *mut c_void,
    out: *mut c_uchar,
    outlen: usize,
    strength: c_uint,
    _prediction_resistance: c_int,
    _adin: *const c_uchar,
    _adin_len: usize,
) -> c_int {
    // SAFETY: `vtest` is a live `ProvTestRng`; `out` is writable for `outlen`.
    unsafe {
        let t = vtest.cast::<ProvTestRng>();
        if strength > (*t).strength {
            return 0;
        }
        if (*t).generate != 0 {
            let mut i: usize = 0;
            while i < outlen {
                *out.add(i) = gen_byte(t);
                i += 1;
            }
        } else {
            if (*t).entropy_len.wrapping_sub((*t).entropy_pos) < outlen {
                return 0;
            }
            ptr::copy_nonoverlapping((*t).entropy.add((*t).entropy_pos), out, outlen);
            (*t).entropy_pos = (*t).entropy_pos.wrapping_add(outlen);
        }
        1
    }
}

/// `static size_t test_rng_nonce(void *vtest, unsigned char *out, unsigned int strength,
/// size_t min_noncelen, size_t max_noncelen)` — `test_rng.c:164-186`.
///
/// **The only `NONCE` entry in the RAND stratum.** It answers `min_noncelen` xorshift bytes in
/// generate mode; otherwise it copies `min(nonce_len, max_noncelen)` from the stored nonce and
/// answers 0 when none was set. In generate mode `out` is written for exactly
/// `min_noncelen` bytes — `max_noncelen` is not consulted.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn test_rng_nonce(
    vtest: *mut c_void,
    out: *mut c_uchar,
    strength: c_uint,
    min_noncelen: usize,
    max_noncelen: usize,
) -> usize {
    // SAFETY: `vtest` is a live `ProvTestRng`; `out` is NULL or writable for the length returned.
    unsafe {
        let t = vtest.cast::<ProvTestRng>();
        if strength > (*t).strength {
            return 0;
        }

        if (*t).generate != 0 {
            let mut i: usize = 0;
            while i < min_noncelen {
                *out.add(i) = gen_byte(t);
                i += 1;
            }
            return min_noncelen;
        }

        if (*t).nonce.is_null() {
            return 0;
        }
        let i = if (*t).nonce_len > max_noncelen {
            max_noncelen
        } else {
            (*t).nonce_len
        };
        if !out.is_null() {
            ptr::copy_nonoverlapping((*t).nonce, out, i);
        }
        i
    }
}

/// The ownership transfer is the contract here: `OSSL_PARAM_get_octet_string` **allocates** the
/// buffer it returns, the old field is `OPENSSL_free`d first, and `entropy_pos` is rewound to
/// 0. A wrong descriptor type is a bare `return 0` with no raise. `nonce` reuses the same
/// `ptr`/`size` temporaries without resetting `ptr` to NULL between the two arms, exactly as the
/// authority writes it — after the entropy arm `ptr` is explicitly re-NULLed, so the nonce arm
/// starts clean.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn test_rng_set_ctx_params(
    vtest: *mut c_void,
    params: *const OsslParam,
) -> c_int {
    // SAFETY: the caller's contract.
    unsafe {
        if vtest.is_null() {
            return 0;
        }
        if let Some(site) = repeated_param_site(params, &TEST_RNG_SET_CTX_PARAMS_DECODER_KEYS) {
            return fail_at(site);
        }

        let t = vtest.cast::<ProvTestRng>();
        let mut ptr: *mut c_void = ptr::null_mut();
        let mut size: usize = 0;

        let p = OSSL_PARAM_locate_const(params, OSSL_RAND_PARAM_STRENGTH);
        if !p.is_null() && OSSL_PARAM_get_uint(p, ptr::addr_of_mut!((*t).strength)) == 0 {
            return 0;
        }

        let p = OSSL_PARAM_locate_const(params, OSSL_RAND_PARAM_TEST_ENTROPY);
        if !p.is_null() {
            if OSSL_PARAM_get_octet_string(p, &mut ptr, 0, &mut size) == 0 {
                return 0;
            }
            CRYPTO_free(
                (*t).entropy.cast(),
                FILE_TEST_RNG,
                LINE_TEST_RNG_SET_FREE_ENTROPY,
            );
            (*t).entropy = ptr.cast::<c_uchar>();
            (*t).entropy_len = size;
            (*t).entropy_pos = 0;
            ptr = ptr::null_mut();
        }

        let p = OSSL_PARAM_locate_const(params, OSSL_RAND_PARAM_TEST_NONCE);
        if !p.is_null() {
            if OSSL_PARAM_get_octet_string(p, &mut ptr, 0, &mut size) == 0 {
                return 0;
            }
            CRYPTO_free(
                (*t).nonce.cast(),
                FILE_TEST_RNG,
                LINE_TEST_RNG_SET_FREE_NONCE,
            );
            (*t).nonce = ptr.cast::<c_uchar>();
            (*t).nonce_len = size;
        }

        let p = OSSL_PARAM_locate_const(params, OSSL_RAND_PARAM_MAX_REQUEST);
        if !p.is_null() && OSSL_PARAM_get_size_t(p, ptr::addr_of_mut!((*t).max_request)) == 0 {
            return 0;
        }

        let p = OSSL_PARAM_locate_const(params, OSSL_RAND_PARAM_GENERATE);
        if !p.is_null() && OSSL_PARAM_get_uint(p, ptr::addr_of_mut!((*t).generate)) == 0 {

        }
        1
    }
}

/// `static int RCT_test(CRNG_TEST *crngt, uint8_t next)` — `fips_crng_test.c:111-131`.
///
/// The Repetition Count Test. The critical values are `rct_c[ENTROPY_H]`, and with the profile's
/// `ENTROPY_H == 6` that is **5** — a run of five identical bytes fails. The first sample of a
/// run is stored with `b = 1`; a *different* sample restarts the run. Note the test passes when
/// `++b < rct_c[H]`, i.e. the *fifth* repeat (b reaches 5) fails.
///
/// # Safety
/// `crngt` is a live `CrngTest`.
unsafe fn RCT_test(crngt: *mut CrngTest, next: u8) -> c_int {
    /*
     * Critical values for this test are computed using:
     *
     *      C = 1 + \left\lceil\frac{ -log_2 \alpha}H\right\rceil
     *
     * where alpha = 2^-20 and H is the expected entropy per sample.
     */
    static RCT_C: [c_uint; 9] = [
        41, /* H = 0.5 */
        21, 11, 8, 6, 5, 5, 4, 4, /* H = 1, ..., 8 */
    ];

    // SAFETY: per the caller's contract.
    unsafe {
        if (*crngt).rct.b != 0 && next == (*crngt).rct.a {
            (*crngt).rct.b = (*crngt).rct.b.wrapping_add(1);
            return c_int::from((*crngt).rct.b < RCT_C[ENTROPY_H]);
        }
        (*crngt).rct.a = next;
        (*crngt).rct.b = 1;
        1
    }
}

/// `static int APT_test(CRNG_TEST *crngt, uint8_t next)` — `fips_crng_test.c:136-163`.
///
/// The Adaptive Proportion Test. With `ENTROPY_H == 6`, `apt_c[6] == 25`: twenty-five
/// occurrences of the same byte in a 512-sample window fails the test, and the window resets on
/// either a failure (`b = 0`) or reaching `ENTROPY_APT_W`. The `> 0`/`i` handling means the
/// run's first sample is *not* counted as an occurrence of a fresh window.
///
/// # Safety
/// `crngt` is a live `CrngTest`.
unsafe fn APT_test(crngt: *mut CrngTest, next: u8) -> c_int {
    /*
     * Critical values for this test are drawn from a binomial distribution with n = 512,
     * p = 2^-H at a critical threshold of 2^-20.  H being the expected entropy per sample.
     * Refer SP 800-90B section 4.4.2, table 2.
     */
    static APT_C: [c_uint; 9] = [
        410, /* H = 0.5 */
        311, 177, 103, 62, 39, 25, 18, 13, /* H = 1, ..., 8 */
    ];

    // SAFETY: per the caller's contract.
    unsafe {
        if (*crngt).apt.b != 0 {
            if (*crngt).apt.a == next {
                (*crngt).apt.b = (*crngt).apt.b.wrapping_add(1);
                if (*crngt).apt.b >= APT_C[ENTROPY_H] {
                    (*crngt).apt.b = 0;
                    return 0;
                }
            }
            (*crngt).apt.i = (*crngt).apt.i.wrapping_add(1);
            if (*crngt).apt.i >= ENTROPY_APT_W {
                (*crngt).apt.b = 0;
            }
            return 1;
        }
        (*crngt).apt.a = next;
        (*crngt).apt.b = 1;
        (*crngt).apt.i = 1;
        1
    }
}

/// `static int crng_test(CRNG_TEST *crngt, const unsigned char *buf, size_t n)` —
/// `fips_crng_test.c:165-177`.
///
/// The two tests run over the same byte in the same pass; a failure of either sets
/// `EVP_RAND_STATE_ERROR` **and** raises `PROV_R_ENTROPY_SOURCE_FAILED_CONTINUOUS_TESTS`, and
/// the whole call fails. The state latch is the contract: once a health test has failed, every
/// later call sees `ERROR` and the caller's own state check refuses it.
///
/// # Safety

unsafe fn crng_test(crngt: *mut CrngTest, buf: *const c_uchar, n: usize) -> c_int {
    // SAFETY: per the caller's contract.
    unsafe {
        let mut i: usize = 0;
        while i < n {
            if RCT_test(crngt, *buf.add(i)) == 0 || APT_test(crngt, *buf.add(i)) == 0 {
                (*crngt).state = EVP_RAND_STATE_ERROR;

                // PROV_R_ENTROPY_SOURCE_FAILED_CONTINUOUS_TESTS).
                raise_site(&err_sites::PROV_FIPS_CRNG_TEST_172);

            }
            i += 1;
        }
        1
    }
}

/// `static const OSSL_DISPATCH *find_call(const OSSL_DISPATCH *dispatch, int function)` —
/// `fips_crng_test.c:179-189`.
///
/// A linear walk for the first entry with `function`, NULL for no match. This is the crate's
/// [`crate::context::dispatch::lookup`] shape; it is kept local because the authority keeps it
/// local and because the caller immediately applies [`entry_function`] to the *entry*.
///
/// # Safety
/// `dispatch` is NULL or an `OSSL_DISPATCH_END`-terminated array that stays live.
unsafe fn find_call(dispatch: *const OsslDispatch, function: c_int) -> *const OsslDispatch {
    if dispatch.is_null() {
        return ptr::null();
    }
    // SAFETY: END-terminated per the caller's contract.
    unsafe {
        let mut p = dispatch;
        while (*p).function_id != OSSL_DISPATCH_END {
            if (*p).function_id == function {
                return p;
            }
            p = p.add(1);
        }
    }
    ptr::null()
}

/// `static void *crng_test_new(void *provctx, void *parent, const OSSL_DISPATCH *p_dispatch)`
/// — `fips_crng_test.c:191-223`.
///
/// The seven parent callbacks are each extracted with a `find_call` + `OSSL_FUNC_...()` cast,
/// and each is optional: a parent that publishes only some of them leaves the rest NULL, and
/// every use site tests for NULL. The child's own state starts `UNINITIALISED`; the health-test
/// structs are zeroed by `zalloc`, which is what `RCT_test`/`APT_test` treat as "no run yet".
///
/// # Safety
/// The dispatch contract; `p_dispatch` is NULL or the parent's terminated table.
unsafe extern "C" fn crng_test_new(
    provctx: *mut c_void,
    parent: *mut c_void,
    p_dispatch: *const OsslDispatch,
) -> *mut c_void {
    // SAFETY: the caller's contract.
    unsafe {
        let crngt = CRYPTO_zalloc(
            core::mem::size_of::<CrngTest>(),
            FILE_FIPS_CRNG_TEST,
            LINE_FIPS_CRNG_TEST_ZALLOC,
        )
        .cast::<CrngTest>();
        if crngt.is_null() {
            return ptr::null_mut();
        }

        (*crngt).provctx = provctx;
        (*crngt).state = EVP_RAND_STATE_UNINITIALISED;

        /* Extract parent's functions */
        if !parent.is_null() {
            (*crngt).parent = parent;

            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_ENABLE_LOCKING);
            if !pfunc.is_null() {
                (*crngt).parent_enable_locking = entry_function::<RandEnableLockingFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_LOCK);
            if !pfunc.is_null() {
                (*crngt).parent_lock = entry_function::<RandLockFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_UNLOCK);
            if !pfunc.is_null() {
                (*crngt).parent_unlock = entry_function::<RandUnlockFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_GETTABLE_CTX_PARAMS);
            if !pfunc.is_null() {
                (*crngt).parent_gettable_ctx_params =
                    entry_function::<RandGettableCtxParamsFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_GET_CTX_PARAMS);
            if !pfunc.is_null() {
                (*crngt).parent_get_ctx_params = entry_function::<RandGetCtxParamsFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_GET_SEED);
            if !pfunc.is_null() {
                (*crngt).parent_get_seed = entry_function::<RandGetSeedFn>(pfunc);
            }
            let pfunc = find_call(p_dispatch, OSSL_FUNC_RAND_CLEAR_SEED);
            if !pfunc.is_null() {

            }
        }

        crngt.cast()

/// `static size_t crng_test_get_seed(void *vcrngt, unsigned char **pout, int entropy,
/// size_t min_len, size_t max_len, int prediction_resistance,
/// const unsigned char *adin, size_t adin_len)` — `fips_crng_test.c:286-321`.
///
/// **The whole unit in one function.** With no parent (or a parent that published no
/// `GET_SEED`), it takes `ossl_prov_get_entropy(provctx, …)` and runs the health tests over the
/// result. With a parent, it takes the parent's lock, calls the parent's `GET_SEED`, runs the
/// tests, and on failure hands the buffer back through the parent's `CLEAR_SEED` if it has one.
/// The lock is released on every path. A failed test answers **0**, not the byte count.
///
/// # Safety
/// The dispatch contract.
unsafe extern "C" fn crng_test_get_seed(
    vcrngt: *mut c_void,
    pout: *mut *mut c_uchar,
    entropy: c_int,
    min_len: usize,
    max_len: usize,
    prediction_resistance: c_int,
    adin: *const c_uchar,
    adin_len: usize,
) -> usize {
    // SAFETY: the caller's contract.
    unsafe {
        let crngt = vcrngt.cast::<CrngTest>();
        let mut r: usize = 0;

        /* Without a parent, we rely on the up calls */
        if (*crngt).parent.is_null() || (*crngt).parent_get_seed.is_none() {
            // MISSING: `ossl_prov_get_entropy` (`providers/.../include/prov/seeding.h:22`) and
            // the `PROV_CTX` type it takes.
            let n = ossl_prov_get_entropy((*crngt).provctx, pout, entropy, min_len, max_len);
            if n == 0 {
                return 0;
            }
            r = crng_test(crngt, *pout, n);
            return if r > 0 { n } else { 0 };
        }

        /* Grab seed from our parent */
        if lock_parent(crngt) == 0 {
            return 0;
        }

        // The guard above proved `parent_get_seed` is `Some`; the `if let` keeps that
        // fact without an unwrapping operation.
        let mut n: usize = 0;
        if let Some(get_seed) = (*crngt).parent_get_seed {
            n = get_seed(
                (*crngt).parent,
                pout,
                entropy,
                min_len,
                max_len,
                prediction_resistance,
                adin,
                adin_len,
            );
        }
        if n > 0 && crng_test(crngt, *pout, n) > 0 {
            r = n;
        } else if let Some(clear) = (*crngt).parent_clear_seed {
            clear((*crngt).parent, *pout, n);
        }
        unlock_parent(crngt);
        r
    }
}

/// `uint32_t ossl_rand_uniform_uint32(OSSL_LIB_CTX *ctx, uint32_t upper, int *err)` —
/// `rand_uniform.c:25-99`.
///
/// An optimal uniform integer in `[0, upper)`: a fixed-point draw on `[0, 1)` is multiplied by
/// `upper`, and the high word is the answer unless the fractional part *might* carry. The early
/// exit is `f <= 1 + ~upper`, which is `upper.wrapping_neg()` — written that way because the
/// C's `~upper` is a `uint32_t` complement and the whole expression wraps.
///
/// Every arithmetic operation here that C performs on unsigned values is transcribed with the
/// `wrapping_*` form so that `overflow-checks = true` cannot turn a defined C wraparound into a
/// panic: `upper * rand` does **not** wrap (it is widened to `uint64_t` first), but `f += f2`
/// and `1 + ~upper` do. `upper == 0` is asserted and answers 0 with `*err = 0` — the only place
/// `err` is cleared; `upper == 1` answers 0 with no error.
///
/// # Safety
/// `err` is writable. `ctx` is NULL or a live library context for `RAND_bytes_ex`.
unsafe extern "C" fn ossl_rand_uniform_uint32(
    ctx: *mut OSSL_LIB_CTX,
    upper: u32,
    err: *mut c_int,
) -> u32 {
    // SAFETY: `err` is writable per the contract.
    unsafe {
        if ossl_assert(upper > 0) == 0 {
            *err = 0;
            return 0;
        }
        if upper == 1 {
            return 0;
        }

        /* Get 32 bits of entropy */
        let mut rand: u32 = 0;
        // MISSING: `RAND_bytes_ex` (`crypto/rand/rand_lib.c`) — Phase 9's own obligation.
        if RAND_bytes_ex(
            ctx,
            ptr::addr_of_mut!(rand).cast::<c_uchar>(),
            core::mem::size_of::<u32>(),
            0,
        ) <= 0
        {
            *err = 1;
            return 0;
        }

        let mut prod: u64 = (upper as u64).wrapping_mul(rand as u64);
        let mut i: u32 = (prod >> 32) as u32;
        let mut f: u32 = (prod & 0xffff_ffff) as u32;
        if f <= upper.wrapping_neg() {
            return i;
        }

        let max_followup_iterations: c_int = 10;
        let mut j: c_int = 0;
        while j < max_followup_iterations {
            // MISSING: `RAND_bytes_ex`.
            if RAND_bytes_ex(
                ctx,
                ptr::addr_of_mut!(rand).cast::<c_uchar>(),
                core::mem::size_of::<u32>(),
                0,
            ) <= 0
            {
                *err = 1;
                return 0;
            }
            prod = (upper as u64).wrapping_mul(rand as u64);
            let f2: u32 = (prod >> 32) as u32;
            f = f.wrapping_add(f2);
            /* On overflow, add the carry to our result */
            if f < f2 {
                return i.wrapping_add(1);
            }
            /* For not all 1 bits, there is no carry so return the result */
            if f != 0xffff_ffff {
                return i;
            }
            /* setup for the next word of randomness */
            f = (prod & 0xffff_ffff) as u32;
            j += 1;
        }
        i
    }
}

/// `uint32_t ossl_rand_range_uint32(OSSL_LIB_CTX *ctx, uint32_t lower, uint32_t upper,
/// int *err)` — `rand_uniform.c:101-109`.
///
/// The refusal is `lower < upper` **asserted**, and `*err = 1` on failure. The subtraction
/// `upper - lower` is unsigned and therefore `wrapping_sub`.
///
/// # Safety
/// `err` is writable; `ctx` is NULL or a live library context.
unsafe extern "C" fn ossl_rand_range_uint32(
    ctx: *mut OSSL_LIB_CTX,
    lower: u32,
    upper: u32,
    err: *mut c_int,
) -> u32 {
    // SAFETY: `err` is writable per the contract.
    unsafe {
        if ossl_assert(lower < upper) == 0 {
            *err = 1;
            return 0;
        }
        lower.wrapping_add(ossl_rand_uniform_uint32(
            ctx,
            upper.wrapping_sub(lower),
            err,
        ))
    }
}

/// The shape the integration must reproduce: `*no_cache` is set to 0 **before** the switch, so
/// the operation tables are cacheable; `OSSL_OP_RAND` selects `base_rands`; and every other
/// operation id — including the four baseprov.c handles (ENCODER, DECODER, STORE, RAND) — falls
/// through to NULL. This function is not provider-row data and is not transcribed as Rust here
/// because the base provider's module does not exist in this crate yet; the row above is the
/// part that belongs to this unit.
///
/// # Safety
/// `no_cache` must be writable; `provctx` is ignored by the RAND arm.
#[allow(dead_code)] // the base provider's module lands separately
unsafe extern "C" fn base_query(
    _provctx: *mut c_void,
    operation_id: c_int,
    no_cache: *mut c_int,
) -> *const OsslAlgorithm {
    // SAFETY: `no_cache` is writable per the contract.
    unsafe { *no_cache = 0 };
    match operation_id {
        OSSL_OP_RAND => BASE_RANDS.as_ptr(),
        _ => ptr::null(),
    }
}
